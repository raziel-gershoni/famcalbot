Settled in this plan (technical, one line each):
- `connectionTimeoutMillis` stays unset (0): the global "no artificial timeouts" rule. Limit: a server that accepts TCP and never answers keeps a model query waiting (5.22 gave up at 5 s); a transaction's start is still bounded by `maxWait` (2 s, P2028, retried).

```ts
// Smoke test for the database client as the app builds it (src/utils/prisma.ts) and the one retry for a database that
// could not be reached (src/utils/db-retry.ts). It runs the real generated client against the fake DATABASE_URL, a refused
// port on 127.0.0.1, so the CI proves on its own Node that the client loads and fails the way withRetry reads.
// No database and no network beyond 127.0.0.1.
//   bash scripts/ci-smokes.sh   (or one smoke with its fake env, see the plan or CLAUDE.md)

import path from 'node:path';

const ROOT = path.resolve(__dirname, '..');
const results: { name: string; ok: boolean; reason?: string }[] = [];
async function check(name: string, fn: () => void | Promise<void>) {
  try {
    await fn();
    results.push({ name, ok: true });
  } catch (err) {
    results.push({ name, ok: false, reason: err instanceof Error ? err.message : String(err) });
  }
}
function assert(cond: unknown, message: string): asserts cond {
  if (!cond) throw new Error(message);
}
const show = (v: unknown) => JSON.stringify(v);

/** An error by its kind only (class, code, Prisma 5's errorCode, the adapter's cause kind): never the message, which names the host. */
function kindOf(error: unknown): string {
  if (!(error instanceof Error)) return `a thrown ${typeof error}`;
  const e = error as { code?: unknown; errorCode?: unknown; meta?: { driverAdapterError?: { cause?: { kind?: unknown } } } };
  return [error.constructor?.name ?? error.name, e.code, e.errorCode, e.meta?.driverAdapterError?.cause?.kind]
    .filter(v => typeof v === 'string').join(' ');
}

/** Calls `fn` through the real withRetry (no wait between attempts) and returns how many times it ran and what was thrown. */
async function callsOf(fn: () => Promise<unknown>): Promise<{ calls: number; thrown: unknown }> {
  const { withRetry } = await import('../src/utils/db-retry');
  let calls = 0;
  let thrown: unknown = null;
  try {
    await withRetry(() => { calls++; return fn(); }, 2, 1);
  } catch (e) { thrown = e; }
  return { calls, thrown };
}

// A check that never settles would let Node drain its event loop and exit 0 in silence: that must read as a failure.
let reported = false;
process.on('exit', () => {
  if (!reported) {
    console.error('FAIL  the smoke ended before its checks did (a promise that never settled)');
    process.exitCode = 1;
  }
});

async function main() {
  // The real client queries whatever DATABASE_URL names: refuse anything but the fake refused port.
  if (process.env.DATABASE_URL !== 'postgresql://fake:fake@127.0.0.1:1/fake') {
    console.error('refusing to run: DATABASE_URL is not the fake refused port of scripts/ci-smokes.sh');
    process.exit(1);
  }
  const { prisma } = await import('../src/utils/prisma');
  // Prisma 7: PrismaClientKnownRequestError P1001. Prisma 5.22: PrismaClientInitializationError, errorCode P1001 on $connect
  // (node24-trial.md §5); a query's implicit connect is read by its words too, in case it carries no errorCode.
  const isP1001 = (e: unknown) => /\bP1001\b/.test(kindOf(e))
    || (e instanceof Error && e.constructor?.name === 'PrismaClientInitializationError' && /Can't reach database server/.test(e.message));

  await check(`the real client loads on Node ${process.version}: a query on a refused port fails as Prisma's "can't reach" (P1001)`, async () => {
    let thrown: unknown = null;
    try { await prisma.user.findFirst({ select: { id: true } }); } catch (e) { thrown = e; }
    assert(isP1001(thrown), `thrown ${kindOf(thrown)}`);
  });

  await check('withRetry retries a query that could not reach the database: 3 calls, then the P1001 reaches the caller', async () => {
    const { calls, thrown } = await callsOf(() => prisma.user.findFirst({ select: { id: true } }));
    assert(calls === 3 && isP1001(thrown), `calls ${calls}, thrown ${kindOf(thrown)}`);
  });

  reported = true;
  let failed = 0;
  for (const r of results) {
    console.log(`${r.ok ? 'OK' : 'FAIL'}  ${r.name}${r.ok ? '' : ` — ${r.reason}`}`);
    if (!r.ok) failed++;
  }
  console.log(`\n${results.length - failed}/${results.length} passed (root ${ROOT})`);
  process.exit(failed ? 1 : 0);
}

main().catch(err => {
  console.error(kindOf(err));
  process.exit(1);
});
```

```ts
  async function droppingServer(): Promise<{ url: string; connections: () => number; close: () => void }> {
```

```ts
/**
 * The one place a database client is built (Prisma 7 with the pg driver adapter), for the app (src/utils/prisma.ts) and
 * the seed. Prisma 7 reads none of the old URL knobs (connection_limit, pool_timeout, connect_timeout): the pool takes
 * its settings here.
 */
import { PrismaClient, type Prisma } from '@prisma/client';
import { PrismaPg } from '@prisma/adapter-pg';
import { captureError } from '../lib/error-capture';

/**
 * - max 5: Prisma 5's default (2 x CPUs + 1) on Railway's 2 vCPUs. Each module instance that loads src/utils/prisma.ts in
 *   production builds its own client and pool (the scheduler and the routes), so the process holds a few times this.
 * - idleTimeoutMillis 60 s: an idle connection closes after a minute, long before Neon suspends (5 idle minutes), and a
 *   burst of requests still reuses its connections (pg's own default, 10 s, would reconnect for nearly every request).
 * - no connectionTimeoutMillis (pg's 0): no timeout of our own. A transaction's start is still bounded by Prisma's
 *   maxWait (2 s), which fails with P2028 "Unable to start a transaction", and withRetry retries it.
 */
export const PRISMA_POOL = { max: 5, idleTimeoutMillis: 60_000 } as const;

/** A pool failure by its kind (class and code): a pg message can name the host. */
export function poolErrorKind(error: Error): string {
  const code = (error as { code?: unknown }).code;
  return [error.constructor?.name ?? error.name, typeof code === 'string' ? code : null].filter(Boolean).join(' ');
}

/**
 * An idle pooled connection that died (no caller saw it: the pool drops it and opens a new one when asked) is a warning;
 * a connection lost inside a transaction is an error (the transaction's caller also gets one).
 */
export function reportPoolError(where: 'idle' | 'transaction', error: Error): void {
  captureError(new Error(`[prisma] ${where} connection failed: ${poolErrorKind(error)}`), 'prisma-pool', undefined, where === 'idle' ? 'warning' : 'error');
}

/** A client on its own pool. Nothing connects until the first query. */
export function createPrismaClient(connectionString: string | undefined, log: Prisma.LogLevel[]): PrismaClient {
  const adapter = new PrismaPg(
    { connectionString, ...PRISMA_POOL },
    { onPoolError: error => reportPoolError('idle', error), onConnectionError: error => reportPoolError('transaction', error) },
  );
  return new PrismaClient({ adapter, log });
}
```

```ts
// Kept apart from prisma.ts so a smoke can load the real retry without a PrismaClient.
import { Prisma } from '@prisma/client';

/** Node's codes for a connect that failed on every address of a host (an AggregateError Prisma does not map). */
const CONNECT_FAILED = new Set(['ECONNREFUSED', 'ETIMEDOUT', 'EHOSTUNREACH', 'ENETUNREACH']);

/**
 * True only when nothing ran: the database could not be reached (P1001 from a model query or a transaction's start; a
 * raw query reports the same as P2010 with the adapter's DatabaseNotReachable; a host whose every address refused gives
 * Node's AggregateError), or a transaction could not start in time (P2028 "Unable to start a transaction"). Not a query
 * or validation error, a refused password (P1000), a lost connection (the statement may have run), or another P2028
 * such as "Transaction already closed". With Prisma 7's adapter nothing connects at startup, so an unreachable database
 * never surfaces as PrismaClientInitializationError.
 */
function nothingRan(error: unknown): boolean {
  if (error instanceof Prisma.PrismaClientKnownRequestError) {
    const kind = (error.meta as { driverAdapterError?: { cause?: { kind?: unknown } } } | undefined)?.driverAdapterError?.cause?.kind;
    return error.code === 'P1001'
      || (error.code === 'P2010' && kind === 'DatabaseNotReachable')
      || (error.code === 'P2028' && error.message.includes('Unable to start a transaction'));
  }
  return error instanceof AggregateError && CONNECT_FAILED.has(String((error as { code?: unknown }).code));
}

/** Retry wrapper for DB operations that may fail while Neon wakes up (see nothingRan). */
export async function withRetry<T>(
  fn: () => Promise<T>,
  maxRetries = 2,
  delayMs = 2000,
): Promise<T> {
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (!nothingRan(error) || attempt === maxRetries) throw error;
      console.warn(`[DB] Connection or transaction start failed (attempt ${attempt + 1}/${maxRetries + 1}), retrying in ${delayMs}ms...`);
      await new Promise(r => setTimeout(r, delayMs));
    }
  }
  throw new Error('Unreachable');
}
```
